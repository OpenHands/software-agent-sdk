"""Disposable live-evidence bundle; identical in local and Docker workspaces."""
from contextlib import closing
import json
import os
from pathlib import Path
import time
from uuid import UUID

from openhands.sdk import RemoteConversation, RemoteWorkspace
from openhands.sdk.conversation.response_utils import get_agent_final_response
from openhands.tools import register_default_tools

register_default_tools()
conversation_id = UUID(os.environ['AUTOMATION_CONVERSATION_ID'])
entrypoint_presence = {name: name in os.environ for name in ('EVIDENCE_ALLOWED', 'EVIDENCE_EXCLUDED')}
print('SCRIPT_SECRET_PRESENCE: ' + json.dumps(entrypoint_presence), flush=True)
print('MASKED_ALLOWED_SECRET: ' + os.environ.get('EVIDENCE_ALLOWED', '<absent>'), flush=True)
print('EVIDENCE_HOLD: 20 seconds for the admission queue observation', flush=True)
time.sleep(20)
workspace_path = Path(os.environ['WORKSPACE_BASE'])
workspace_path.mkdir(parents=True, exist_ok=True)
(workspace_path / 'entrypoint.json').write_text(json.dumps(entrypoint_presence))
with RemoteWorkspace(
    host=os.environ['AGENT_SERVER_URL'],
    api_key=os.environ['SESSION_API_KEY'],
    working_dir=str(workspace_path),
) as workspace, closing(RemoteConversation.attach(workspace, conversation_id)) as conversation:
    conversation.send_message(
        'This is a disposable infrastructure validation workspace. First use the terminal '
        'to read entrypoint.json and explain those observed script-entrypoint booleans. '
        'Then use the terminal tool '
        'to run Python that checks whether environment variables EVIDENCE_ALLOWED and '
        'EVIDENCE_EXCLUDED exist, reporting only boolean presence and never their values. '
        'Write the two booleans to fixture.json in the current workspace, print the file, '
        'and finish with a short sentence explaining the observed booleans. '
        'Do not access the network, repositories, or any other credentials.'
    )
    conversation.run(timeout=180)
    print('AGENT_FINISHED: ' + get_agent_final_response(conversation.state.events), flush=True)
print('EVIDENCE_INSPECT: 25 seconds for the real Canvas file and terminal views', flush=True)
time.sleep(25)
print('EVIDENCE_BUNDLE_COMPLETED', flush=True)
